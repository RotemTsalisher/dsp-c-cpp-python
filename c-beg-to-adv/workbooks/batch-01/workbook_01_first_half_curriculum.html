Exercise 1 — Bench warm-up (advanced)

Context: A field tech needs a quick ASCII scope trace of one period of a comb-delay test tone before the uplink chain is enabled.

Write a program that uses variables int samples, double amplitude, double frequency_hz, and double sample_rate_hz. For each sample index n from 0 to samples - 1, compute x = amplitude * sin(2*pi*frequency_hz*n/sample_rate_hz) (use math.h) and draw one row: a centered vertical bar whose height is proportional to |x| (map to 0–20 * characters, pad with spaces). Print the numeric value with printf using %6.3f beside the bar.

Topics: drawing shapes with prints; variables; printf format specifiers; working with numbers.

/* gcc -std=c11 -Wall -Wextra -O2 main.c -lm */
#include <math.h>
#include <stdio.h>

int main(void)
{
    int const samples = 32;
    double const amplitude = 0.85;
    double const frequency_hz = 440.0;
    double const sample_rate_hz = 48000.0;
    double const two_pi = 2.0 * 3.14159265358979323846;

    printf("=== Comb-line tone scope (1 period) ===\n");
    for (int n = 0; n < samples; ++n) {
        double const t = (double)n / sample_rate_hz;
        double const x = amplitude * sin(two_pi * frequency_hz * t);
        int const bar_len = (int)(fabs(x) * 20.0 / amplitude);
        printf("%6.3f |", x);
        for (int s = 0; s < 20 - bar_len; ++s) {
            printf(" ");
        }
        for (int b = 0; b < bar_len; ++b) {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}
Why this fits the curriculum: Variables hold physical units (Hz, amplitude). printf combines a fixed-width float field with hand-built ASCII graphics — the “drawing shapes” skill applied to a real bench readout, not toy art.
Exercise 2

Context: Dual ADC channels feed a mono-mix front-end. The log must show raw counts, converted volts, and a channel tag.

Declare int counts_left, int counts_right, char channel_tag ('L' or 'R' for the row being printed), and double volts. Given int resolution_bits = 12 and double full_scale_volts = 2.5, print one formatted line per channel:

CH %c | counts=%04d | V=%7.4f

Use the correct conversion: volts = (counts / (2^bits - 1)) * full_scale with double arithmetic (promote before divide). Demonstrate both channels in main.

Topics: data types (int, double, char); printf specifiers %c, %04d, %7.4f; numeric promotion.

#include <math.h>
#include <stdio.h>

int main(void)
{
    int const resolution_bits = 12;
    double const full_scale_volts = 2.5;
    double const denom = pow(2.0, (double)resolution_bits) - 1.0;

    int const counts_left = 2047;
    int const counts_right = 1024;

    printf("=== AFE dual-channel snapshot ===\n");
    {
        char const tag = 'L';
        int const counts = counts_left;
        double const volts = ((double)counts / denom) * full_scale_volts;
        printf("CH %c | counts=%04d | V=%7.4f\n", tag, counts, volts);
    }
    {
        char const tag = 'R';
        int const counts = counts_right;
        double const volts = ((double)counts / denom) * full_scale_volts;
        printf("CH %c | counts=%04d | V=%7.4f\n", tag, counts, volts);
    }
    return 0;
}
Why this fits the curriculum: Mixing int counts with double full-scale requires explicit promotion — a common EE bug if left implicit. Width/precision specifiers keep columns aligned for HIL logs.
Exercise 3

Context: Uplink mono-mix and PSD reference level for a correlated stereo tone.

Given double left and double right (linear samples), compute:

  • mono = 0.5 * (left + right) (comb-safe downmix)
  • quad_energy = 0.5 * (left*left + right*right)
  • db10 = 10.0 * log10(quad_energy) when quad_energy > 0, else print -inf as text

Store the logic in main only (functions are a later subject). Print all three values with printf. Test with left = right = 0.25 and with left = 0.3, right = -0.3.

Topics: working with numbers; float/double math; printf for mixed outputs.

#include <math.h>
#include <stdio.h>

int main(void)
{
    printf("=== Mono-mix / PSD reference ===\n");

    for (int case_id = 0; case_id < 2; ++case_id) {
        double left = 0.25;
        double right = 0.25;
        if (case_id == 1) {
            left = 0.3;
            right = -0.3;
        }
        double const mono = 0.5 * (left + right);
        double const quad_energy = 0.5 * (left * left + right * right);
        printf("L=% .3f R=% .3f -> mono=% .4f energy=% .6f ",
               left, right, mono, quad_energy);
        if (quad_energy > 0.0) {
            double const db10 = 10.0 * log10(quad_energy);
            printf("PSD_ref=% .2f dB\n", db10);
        } else {
            printf("PSD_ref=-inf dB\n");
        }
    }
    return 0;
}
Why this fits the curriculum: The 0.5 factors encode EE semantics (mono mix + mean-square energy). log10 from math.h is the minimal math extension for dB readouts at this course stage.
Exercise 4

Context: Firmware headers document immutable calibration constants for an AFE register block.

Define:

  • #define AFE_ADC_BITS 12
  • const double AFE_FULL_SCALE_V = 2.5;
  • const float COMB_FEEDBACK_MAX = 0.95f;

Print a three-row table from main (no functions yet — subject 12 comes later). Compute and print counts_per_volt = (2^AFE_ADC_BITS - 1) / AFE_FULL_SCALE_V as a double derived constant.

Topics: constants (#define and const); data types; formatted output.

#include <math.h>
#include <stdio.h>

#define AFE_ADC_BITS 12

static double const AFE_FULL_SCALE_V = 2.5;
static float const COMB_FEEDBACK_MAX = 0.95f;

int main(void)
{
    double const counts_per_volt =
        (pow(2.0, (double)AFE_ADC_BITS) - 1.0) / AFE_FULL_SCALE_V;

    printf("=== AFE calibration constants ===\n");
    printf("%-22s %8d  (int/#define)\n", "AFE_ADC_BITS", AFE_ADC_BITS);
    printf("%-22s %8.4lf (double const)\n", "AFE_FULL_SCALE_V", AFE_FULL_SCALE_V);
    printf("%-22s %8.4f  (float const)\n", "COMB_FEEDBACK_MAX", COMB_FEEDBACK_MAX);
    printf("%-22s %8.2lf (derived)\n", "counts_per_volt", counts_per_volt);
    return 0;
}
Why this fits the curriculum: Splitting macro vs const mirrors real headers: macros for compile-time integers, typed const for physical units. Derived constants stay in double to avoid precision loss in PSD scaling.
Exercise 5

Context: Bring-up script dumps the first four AFE shadow registers as a hex table with decoded fields.

Use unsigned int values 0x00000A01, 0x80001234, 0x000000FF, 0xDEAD0000. Print a header row, then each row as:

REG[%d] raw=0x%08X low_byte=0x%02X msb_set=%c

where low_byte is raw & 0xFF and msb_set is 'Y' if bit 31 is set else 'N'. Right-align columns. No arrays yet — unroll four calls or use four labeled variables.

Topics: integers and hex formatting (%X, %02X); char flags; variables; ASCII table drawing.

#include <stdio.h>

int main(void)
{
    unsigned int const reg0 = 0x00000A01u;
    unsigned int const reg1 = 0x80001234u;
    unsigned int const reg2 = 0x000000FFu;
    unsigned int const reg3 = 0xDEAD0000u;

    printf("=== AFE shadow register map ===\n");
    printf("IDX      RAW        LOW   MSB\n");

    for (int index = 0; index < 4; ++index) {
        unsigned int raw = reg0;
        if (index == 1) raw = reg1;
        else if (index == 2) raw = reg2;
        else if (index == 3) raw = reg3;
        unsigned int const low_byte = raw & 0xFFu;
        char const msb_set = (raw & 0x80000000u) ? 'Y' : 'N';
        printf("REG[%d]  raw=0x%08X  low_byte=0x%02X  msb_set=%c\n",
               index, raw, low_byte, msb_set);
    }
    return 0;
}
Why this fits the curriculum: Hex printf specifiers are mandatory for register work. Bit tests return char markers — a bridge between integer math and human-readable logs.
Exercise 6 — System panel

Context: Offline PSD sanity check — eight narrowband bins must be sketched as an ASCII histogram before FFT firmware ships.

Store eight double bin powers in individual variables b0…b7 (use values like 0.02, 0.05, 0.11, 0.40, 0.22, 0.08, 0.03, 0.01). Find max_power with a chain of comparisons (no arrays). For each bin, print:

bin %d | %5.3f | <bar> where bar length is (int)(power / max_power * 30) asterisks. Print max_power and total energy sum(b0..b7) after the chart.

Topics: drawing with prints; variables; numbers; constants for bin count; printf alignment.

#include <stdio.h>

int main(void)
{
    int const bin_count = 8;
    double const b0 = 0.02, b1 = 0.05, b2 = 0.11, b3 = 0.40;
    double const b4 = 0.22, b5 = 0.08, b6 = 0.03, b7 = 0.01;

    double max_power = b0;
    if (b1 > max_power) max_power = b1;
    if (b2 > max_power) max_power = b2;
    if (b3 > max_power) max_power = b3;
    if (b4 > max_power) max_power = b4;
    if (b5 > max_power) max_power = b5;
    if (b6 > max_power) max_power = b6;
    if (b7 > max_power) max_power = b7;

    double const total = b0 + b1 + b2 + b3 + b4 + b5 + b6 + b7;

    printf("=== PSD bin histogram (pre-FFT) ===\n");
    for (int k = 0; k < bin_count; ++k) {
        double power = 0.0;
        if (k == 0) power = b0;
        else if (k == 1) power = b1;
        else if (k == 2) power = b2;
        else if (k == 3) power = b3;
        else if (k == 4) power = b4;
        else if (k == 5) power = b5;
        else if (k == 6) power = b6;
        else power = b7;
        int const bar_len = (int)(power / max_power * 30.0);
        printf("bin %d | %5.3f | ", k, power);
        for (int i = 0; i < bar_len; ++i) {
            putchar('*');
        }
        putchar('\n');
    }
    printf("max_power=%.4f  total_energy=%.4f\n", max_power, total);
    return 0;
}
Why this fits the curriculum: Eight named double variables plus an index loop avoid static arrays (subject 10) while still building a histogram. Normalizing by max_power is core PSD display logic.
Exercise 7 — Half capstone

Context: Comb-line bench report — one screen summarizing delay metadata, mono-mix headroom, and a mini spectrum sketch for factory QA.

Using only topics from subjects 1–6, build one main that:

  1. Declares const / #define constants: SAMPLE_RATE_HZ, COMB_DELAY_MS, ADC_BITS, FULL_SCALE_V.
  2. Computes delay_samples = (COMB_DELAY_MS / 1000.0) * SAMPLE_RATE_HZ as double.
  3. Reads hard-coded int counts for L/R, converts to volts, mono-mixes, prints headroom to full scale in percent.
  4. Draws a 5-bin ASCII PSD strip (variables p0…p4) normalized to the peak.
  5. Prints a boxed footer using only printf (e.g. a line of = characters).

Topics: all six — drawing, variables, types, printf, numbers, constants.

#include <math.h>
#include <stdio.h>

#define SAMPLE_RATE_HZ 48000
#define COMB_DELAY_MS 2.5
#define ADC_BITS 12

static double const FULL_SCALE_V = 2.5;

int main(void)
{
    int const counts_left = 1800;
    int const counts_right = 2200;

    double const denom = pow(2.0, (double)ADC_BITS) - 1.0;
    double const delay_samples = (COMB_DELAY_MS / 1000.0) * (double)SAMPLE_RATE_HZ;
    double const volts_left = ((double)counts_left / denom) * FULL_SCALE_V;
    double const volts_right = ((double)counts_right / denom) * FULL_SCALE_V;
    double const mono = 0.5 * (volts_left + volts_right);
    double const headroom_pct = 100.0 * (1.0 - fabs(mono) / FULL_SCALE_V);

    double const p0 = 0.04, p1 = 0.09, p2 = 0.35, p3 = 0.18, p4 = 0.06;
    double peak = p0;
    if (p1 > peak) peak = p1;
    if (p2 > peak) peak = p2;
    if (p3 > peak) peak = p3;
    if (p4 > peak) peak = p4;

    printf("========================================\n");
    printf(" CardoLink AFE — Comb-line QA panel\n");
    printf("========================================\n");
    printf("Delay: %.2f ms -> %.1f samples @ %d Hz\n",
           COMB_DELAY_MS, delay_samples, SAMPLE_RATE_HZ);
    printf("ADC: L=%04d R=%04d | mono=%7.4f V | headroom=%5.1f%%\n",
           counts_left, counts_right, mono, headroom_pct);
    printf("PSD strip (normalized):\n");

    for (int k = 0; k < 5; ++k) {
        double power = 0.0;
        if (k == 0) power = p0;
        else if (k == 1) power = p1;
        else if (k == 2) power = p2;
        else if (k == 3) power = p3;
        else power = p4;
        int const n = (int)(power / peak * 24.0);
        printf(" %d |%5.3f| ", k, power);
        for (int i = 0; i < n; ++i) {
            putchar('#');
        }
        putchar('\n');
    }
    printf("========================================\n");
    return 0;
}
Why this fits the curriculum: The capstone mirrors a single QA terminal screen: constants define the hardware, numeric promotion handles ADC math, mono-mix checks uplink headroom, and ASCII graphics communicate PSD shape — all with first-half C only.
