Exercise 1 — Bench session capture (advanced)

Context: Before a comb-filter HIL run, the operator must log session metadata — including a free-text bench location that may contain spaces.

Prompt and read with scanf: int sample_rate_hz, double comb_delay_ms, double feedback_gain, and char uplink_mode ('M' mono-mix or 'S' stereo passthrough). Flush the leftover newline, then read a line with fgets into char bench_location[64] (strip trailing \n). Compute delay_samples = (comb_delay_ms / 1000.0) * sample_rate_hz and print a formatted session card.

Topics: user inputs — scanf for numeric/char types; fgets for strings with whitespace.

/* gcc -std=c11 -Wall -Wextra -O2 main.c -lm */
#include <stdio.h>

static void trim_newline(char *s)
{
    if (s == NULL) {
        return;
    }
    for (int i = 0; s[i] != '\0'; ++i) {
        if (s[i] == '\n') {
            s[i] = '\0';
            break;
        }
    }
}

int main(void)
{
    int sample_rate_hz = 0;
    double comb_delay_ms = 0.0;
    double feedback_gain = 0.0;
    char uplink_mode = '?';
    char bench_location[64] = {0};

    printf("=== Comb-filter HIL session log ===\n");
    printf("Sample rate (Hz): ");
    scanf("%d", &sample_rate_hz);
    printf("Comb delay (ms): ");
    scanf("%lf", &comb_delay_ms);
    printf("Feedback gain (0..1): ");
    scanf("%lf", &feedback_gain);
    printf("Uplink mode (M=mono / S=stereo): ");
    scanf(" %c", &uplink_mode);
    (void)getchar();
    printf("Bench location (free text): ");
    if (fgets(bench_location, (int)sizeof bench_location, stdin) == NULL) {
        bench_location[0] = '\0';
    }
    trim_newline(bench_location);

    double const delay_samples =
        (comb_delay_ms / 1000.0) * (double)sample_rate_hz;

    printf("\n--- Session card ---\n");
    printf("Location : %s\n", bench_location);
    printf("Fs       : %d Hz\n", sample_rate_hz);
    printf("Delay    : %.3f ms -> %.2f samples\n",
           comb_delay_ms, delay_samples);
    printf("Feedback : %.4f\n", feedback_gain);
    printf("Uplink   : %c\n", uplink_mode);
    return 0;
}
Why this fits the curriculum: Mixing scanf and fgets is the classic C input pitfall — flushing the newline before reading a spaced string mirrors real bench scripts. Delay-sample math reuses first-half numeric skills without repeating the workbook 01 scope trace.
Exercise 2

Context: A comb delay line holds the last eight mono-mix samples in a circular buffer for notch visualization.

Declare static double delay_ring[8] pre-filled with sample values. Write void print_ring(double const *buf, int len) (no return) that prints each slot as [i] %8.4f. Write double ring_energy(double const *buf, int len) returning mean-square energy sum(x*x)/len. Call both from main and print energy in dB with 10*log10 when > 0.

Topics: static arrays (double); functions with parameters and typed return; math.h.

#include <math.h>
#include <stdio.h>

static void print_ring(double const *buf, int len)
{
    for (int i = 0; i < len; ++i) {
        printf("[%d] %8.4f\n", i, buf[i]);
    }
}

static double ring_energy(double const *buf, int len)
{
    double sum = 0.0;
    for (int i = 0; i < len; ++i) {
        sum += buf[i] * buf[i];
    }
    return sum / (double)len;
}

int main(void)
{
    static double delay_ring[8] = {
        0.12, -0.08, 0.05, 0.31, -0.22, 0.17, -0.04, 0.09
    };
    int const len = 8;

    printf("=== Comb delay ring snapshot ===\n");
    print_ring(delay_ring, len);

    double const energy = ring_energy(delay_ring, len);
    printf("Mean-square energy: %.6f\n", energy);
    if (energy > 0.0) {
        printf("PSD level: %.2f dB\n", 10.0 * log10(energy));
    } else {
        printf("PSD level: -inf dB\n");
    }
    return 0;
}
Why this fits the curriculum: A fixed-size ring models on-chip delay memory — array indexing plus small functions separate display (void) from measurement (typed return). Unlike workbook 01’s eight named variables, the buffer is a first-class static array.
Exercise 3

Context: AFE driver helpers convert raw counts, downmix to mono, and flag clipping before uplink.

Implement these functions (use math.h where needed):

  • double counts_to_volts(int counts, int bits, double full_scale_v)
  • double mono_mix(double left, double right) — comb-safe 0.5*(L+R)
  • int is_clipping(double sample, double full_scale_v) — returns 1 if |sample| >= 0.98 * full_scale, else 0
  • void print_afe_row(char tag, int counts, double volts, int clip)

From main, process two hard-coded channel count pairs and print a two-row table plus the mixed output line.

Topics: functions — parameters, void return, typed returns, math conversion.

#include <math.h>
#include <stdio.h>

static double counts_to_volts(int counts, int bits, double full_scale_v)
{
    double const denom = pow(2.0, (double)bits) - 1.0;
    return ((double)counts / denom) * full_scale_v;
}

static double mono_mix(double left, double right)
{
    return 0.5 * (left + right);
}

static int is_clipping(double sample, double full_scale_v)
{
    return (fabs(sample) >= 0.98 * full_scale_v) ? 1 : 0;
}

static void print_afe_row(char tag, int counts, double volts, int clip)
{
    printf("CH %c | counts=%04d | V=%7.4f | clip=%s\n",
           tag, counts, volts, clip ? "YES" : "no");
}

int main(void)
{
    int const bits = 12;
    double const fs_v = 2.5;
    int const counts_l = 3950;
    int const counts_r = 4010;

    double const v_l = counts_to_volts(counts_l, bits, fs_v);
    double const v_r = counts_to_volts(counts_r, bits, fs_v);
    double const mono = mono_mix(v_l, v_r);

    printf("=== AFE mono-mix clip guard ===\n");
    print_afe_row('L', counts_l, v_l, is_clipping(v_l, fs_v));
    print_afe_row('R', counts_r, v_r, is_clipping(v_r, fs_v));
    printf("MONO mix: %7.4f V | clip=%s\n",
           mono, is_clipping(mono, fs_v) ? "YES" : "no");
    return 0;
}
Why this fits the curriculum: Decomposing ADC math into reusable functions is how firmware headers grow — each helper has a single EE meaning. The clip threshold uses a ratio constant, not the dual-channel snapshot duplicated in workbook 01.
Exercise 4

Context: A technician types measured PSD bin powers from a spectrum analyzer into firmware for a quick uplink sanity plot.

Declare static double psd_bins[16]. Ask the user for int bin_count (1–16), then read that many powers with scanf("%lf", ...). Implement double find_peak(double const *arr, int n) and void draw_psd_bars(double const *arr, int n, double peak) (ASCII bars scaled to 28 chars). Print peak and total energy after the chart.

Topics: static arrays with user inputs; functions; printf drawing (subjects 1–6 cumulative).

#include <stdio.h>

static double find_peak(double const *arr, int n)
{
    double peak = arr[0];
    for (int i = 1; i < n; ++i) {
        if (arr[i] > peak) {
            peak = arr[i];
        }
    }
    return peak;
}

static void draw_psd_bars(double const *arr, int n, double peak)
{
    for (int k = 0; k < n; ++k) {
        int const bar_len = (peak > 0.0) ? (int)(arr[k] / peak * 28.0) : 0;
        printf("bin %2d | %7.4f | ", k, arr[k]);
        for (int i = 0; i < bar_len; ++i) {
            putchar('#');
        }
        putchar('\n');
    }
}

int main(void)
{
    static double psd_bins[16];
    int bin_count = 0;

    printf("=== Manual PSD bin entry ===\n");
    printf("How many bins (1-16)? ");
    scanf("%d", &bin_count);
    if (bin_count < 1) {
        bin_count = 1;
    }
    if (bin_count > 16) {
        bin_count = 16;
    }

    for (int k = 0; k < bin_count; ++k) {
        printf("Bin %d power: ", k);
        scanf("%lf", &psd_bins[k]);
    }

    double const peak = find_peak(psd_bins, bin_count);
    double total = 0.0;
    for (int k = 0; k < bin_count; ++k) {
        total += psd_bins[k];
    }

    printf("\n--- Normalized strip ---\n");
    draw_psd_bars(psd_bins, bin_count, peak);
    printf("peak=%.4f  total=%.4f\n", peak, total);
    return 0;
}
Why this fits the curriculum: Subject 11 pairs array storage with interactive fill — the bounded 16-slot buffer matches analyzer export limits. Peak-normalized bars reuse drawing skills from the first half but driven by runtime data, not hard-coded variables.
Exercise 5

Context: AFE register documentation uses parallel string labels and integer shadow values; one register may be patched at bring-up.

Declare static char reg_names[5][20] and static unsigned int reg_values[5] with meaningful comb/AFE names and sample hex values. Write void print_reg_table(char names[][20], unsigned int const *vals, int n) and unsigned int decode_gain_field(unsigned int raw) returning (raw >> 8) & 0xFF. Prompt the user for a register index (0–4) and a new hex value (%x), update the array, then reprint the table with decoded gain fields.

Topics: static arrays (char strings + unsigned int); user input; functions.

#include <stdio.h>

static unsigned int decode_gain_field(unsigned int raw);

static void print_reg_table(char names[][20], unsigned int const *vals, int n)
{
    printf("%-18s  RAW        GAIN_FLD\n", "REGISTER");
    for (int i = 0; i < n; ++i) {
        unsigned int const gain = decode_gain_field(vals[i]);
        printf("%-18s  0x%08X  0x%02X\n", names[i], vals[i], gain);
    }
}

static unsigned int decode_gain_field(unsigned int raw)
{
    return (raw >> 8) & 0xFFu;
}

int main(void)
{
    static char reg_names[5][20] = {
        "COMB_DELAY",
        "COMB_FB",
        "MONO_MIX",
        "PSD_WINDOW",
        "AFE_CTRL"
    };
    static unsigned int reg_values[5] = {
        0x000003C0u,
        0x0000F333u,
        0x00008000u,
        0x00000400u,
        0x80000001u
    };
    int const n = 5;
    int index = 0;
    unsigned int patch = 0u;

    printf("=== AFE shadow register editor ===\n");
    print_reg_table(reg_names, reg_values, n);

    printf("\nPatch register index (0-4): ");
    scanf("%d", &index);
    if (index < 0) {
        index = 0;
    }
    if (index >= n) {
        index = n - 1;
    }
    printf("New hex value (e.g. DEADBEEF): ");
    scanf("%x", &patch);
    reg_values[index] = patch;

    printf("\n--- After patch ---\n");
    print_reg_table(reg_names, reg_values, n);
    return 0;
}
Why this fits the curriculum: Parallel string and integer arrays model register-map headers. User patching exercises in-place array update plus hex scanf — distinct from workbook 01’s read-only four-register dump.
Exercise 6 — Milestone project (1)

Context: Factory AFE bring-up console — one interactive program that walks an operator through capture, conversion, mono-mix headroom check, and a live PSD strip sketch.

Build a single program combining subjects 1–12:

  1. fgets the operator name (spaces allowed) and scanf for adc_bits, full_scale_v, and L/R counts.
  2. Store five PSD reference bins in static double ref_psd[5]; let the user overwrite each bin.
  3. Functions: counts_to_volts, mono_mix, headroom_pct, draw_psd_strip.
  4. Print a boxed report: operator, volts, mono, headroom %, ASCII PSD strip normalized to peak, footer line of = characters.

Topics: milestone (1) project — all knowledge through subject 12; cumulative first-half skills.

#include <math.h>
#include <stdio.h>

static void trim_newline(char *s)
{
    if (s == NULL) {
        return;
    }
    for (int i = 0; s[i] != '\0'; ++i) {
        if (s[i] == '\n') {
            s[i] = '\0';
            break;
        }
    }
}

static double counts_to_volts(int counts, int bits, double full_scale_v)
{
    double const denom = pow(2.0, (double)bits) - 1.0;
    return ((double)counts / denom) * full_scale_v;
}

static double mono_mix(double left, double right)
{
    return 0.5 * (left + right);
}

static double headroom_pct(double sample, double full_scale_v)
{
    return 100.0 * (1.0 - fabs(sample) / full_scale_v);
}

static double find_peak5(double const *arr)
{
    double peak = arr[0];
    for (int i = 1; i < 5; ++i) {
        if (arr[i] > peak) {
            peak = arr[i];
        }
    }
    return peak;
}

static void draw_psd_strip(double const *arr, double peak)
{
    for (int k = 0; k < 5; ++k) {
        int const n = (peak > 0.0) ? (int)(arr[k] / peak * 22.0) : 0;
        printf(" %d |%6.4f| ", k, arr[k]);
        for (int i = 0; i < n; ++i) {
            putchar('*');
        }
        putchar('\n');
    }
}

int main(void)
{
    char operator_name[48] = {0};
    int adc_bits = 12;
    double full_scale_v = 2.5;
    int counts_l = 0;
    int counts_r = 0;
    static double ref_psd[5] = {0.06, 0.14, 0.38, 0.20, 0.07};

    printf("========================================\n");
    printf(" AFE Bring-up Console — Milestone (1)\n");
    printf("========================================\n");

    printf("Operator name: ");
    if (fgets(operator_name, (int)sizeof operator_name, stdin) == NULL) {
        operator_name[0] = '\0';
    }
    trim_newline(operator_name);

    printf("ADC bits: ");
    scanf("%d", &adc_bits);
    printf("Full scale (V): ");
    scanf("%lf", &full_scale_v);
    printf("Counts L: ");
    scanf("%d", &counts_l);
    printf("Counts R: ");
    scanf("%d", &counts_r);

    printf("\nEnter 5 PSD reference bin powers:\n");
    for (int k = 0; k < 5; ++k) {
        printf("  ref_psd[%d]: ", k);
        scanf("%lf", &ref_psd[k]);
    }

    double const v_l = counts_to_volts(counts_l, adc_bits, full_scale_v);
    double const v_r = counts_to_volts(counts_r, adc_bits, full_scale_v);
    double const mono = mono_mix(v_l, v_r);
    double const headroom = headroom_pct(mono, full_scale_v);
    double const peak = find_peak5(ref_psd);

    printf("\n========================================\n");
    printf("Operator : %s\n", operator_name);
    printf("ADC L/R  : %04d / %04d  -> %.4f / %.4f V\n",
           counts_l, counts_r, v_l, v_r);
    printf("Mono mix : %.4f V  |  headroom: %.1f%%\n", mono, headroom);
    printf("PSD reference strip:\n");
    draw_psd_strip(ref_psd, peak);
    printf("========================================\n");
    return 0;
}
Why this fits the curriculum: Milestone (1) demands a coherent workflow — not a snippet. Inputs, arrays, functions, formatted drawing, and physical-unit math appear in one factory-style screen, extending (not copying) the first-half capstone by making every stage interactive.
Exercise 7 — Milestone game (2)

Context: Training game PSD Peak Hunt — the hidden loudest bin in a comb-notch spectrum must be found in limited guesses; score tracks how quickly trainees locate uplink interference.

Implement a turn-based console game using subjects 1–12:

  1. Seed a static double spectrum[8] with a hidden peak (hard-code or pick index 5 with value 0.85, others lower).
  2. fgets the player name; scanf each guess as a bin index.
  3. Functions: double bin_power(int k), const char *hint_for(double guess_pwr, double peak_pwr), void print_board(double const *spec, int n, int revealed_peak), int update_score(int score, int turns_left).
  4. Five guesses max; hints: "COLD" / "WARM" / "HOT" by power ratio; reveal chart and final score at end.

Topics: milestone (2) small game; arrays, inputs, functions, ASCII display.

#include <stdio.h>

static void trim_newline(char *s)
{
    if (s == NULL) {
        return;
    }
    for (int i = 0; s[i] != '\0'; ++i) {
        if (s[i] == '\n') {
            s[i] = '\0';
            break;
        }
    }
}

static double bin_power(int k)
{
    static double const spectrum[8] = {
        0.04, 0.07, 0.11, 0.19, 0.28, 0.85, 0.16, 0.05
    };
    if (k < 0 || k >= 8) {
        return 0.0;
    }
    return spectrum[k];
}

static int peak_index(void)
{
    return 5;
}

static char const *hint_for(double guess_pwr, double peak_pwr)
{
    if (peak_pwr <= 0.0) {
        return "COLD";
    }
    double const ratio = guess_pwr / peak_pwr;
    if (ratio >= 0.75) {
        return "HOT";
    }
    if (ratio >= 0.35) {
        return "WARM";
    }
    return "COLD";
}

static void print_board(double const *spec, int n, int revealed_peak)
{
    double peak = spec[0];
    for (int i = 1; i < n; ++i) {
        if (spec[i] > peak) {
            peak = spec[i];
        }
    }
    for (int k = 0; k < n; ++k) {
        int const bar = (peak > 0.0) ? (int)(spec[k] / peak * 18.0) : 0;
        printf("%d |%5.2f| ", k, spec[k]);
        for (int i = 0; i < bar; ++i) {
            putchar(revealed_peak ? '#' : '?');
        }
        putchar('\n');
    }
}

static int update_score(int score, int turns_left)
{
    return score + turns_left * 10;
}

int main(void)
{
    static double spectrum[8] = {
        0.04, 0.07, 0.11, 0.19, 0.28, 0.85, 0.16, 0.05
    };
    char player[40] = {0};
    int const max_turns = 5;
    int turns_left = max_turns;
    int score = 0;
    int won = 0;
    int const answer = peak_index();
    double const peak_pwr = spectrum[answer];

    printf("=== PSD Peak Hunt (training) ===\n");
    printf("Find the hidden comb-notch peak bin (0-7).\n");
    printf("Player name: ");
    if (fgets(player, (int)sizeof player, stdin) == NULL) {
        player[0] = '\0';
    }
    trim_newline(player);

    while (turns_left > 0 && !won) {
        int guess = -1;
        printf("\nGuesses left: %d  |  Enter bin index: ", turns_left);
        if (scanf("%d", &guess) != 1) {
            break;
        }
        if (guess < 0 || guess >= 8) {
            printf("Out of range — try 0..7.\n");
            continue;
        }

        double const pwr = bin_power(guess);
        printf("Bin %d power=%.2f  hint: %s\n",
               guess, pwr, hint_for(pwr, peak_pwr));

        if (guess == answer) {
            won = 1;
            score = update_score(score, turns_left);
            printf("Correct! Interference bin located.\n");
        }
        --turns_left;
    }

    printf("\n--- Final spectrum ---\n");
    print_board(spectrum, 8, 1);
    printf("\nPlayer: %s  |  Score: %d  |  %s\n",
           player, score, won ? "PASS" : "RETRY");
    return 0;
}
Why this fits the curriculum: Milestone (2) turns the same C toolkit into engagement — arrays hold the hidden spectrum, scanf/fgets drive rounds, and helper functions keep game logic readable. The masked-then-revealed ASCII chart reinforces PSD literacy under time pressure.
