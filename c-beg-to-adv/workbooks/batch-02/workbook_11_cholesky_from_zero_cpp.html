A. The problem: solve a linear system

You have a matrix R (size L×L) and a vector p (length L). You want vector h such that:

R · h = p

In school this is “simultaneous equations.” In DSP (Wiener–Hopf FIR) h are the FIR taps.

Cholesky does not invent the filter. It only solves the system once R and p exist.

עברית

זו מערכת משוואות ליניאריות. ב־WH ל־FIR, h הם מקדמי הפילטר. Cholesky לא בונה את המסכה — רק פותר.

B. When Cholesky is allowed

Cholesky needs R to be:

  • Symmetric (real): R[i][j] = R[j][i]
  • Positive-definite (PD): all eigenvalues > 0, or equivalently xᵀ R x > 0 for all x≠0

Autocorrelation / Gram matrices from least-squares are typically SPD (or made so with a tiny diagonal regularizer ε).

עברית

צריך מטריצה סימטרית וחיובית־מוגדרת. מטריצות אוטוקורלציה / Gram בדרך כלל כאלה — ולפעמים מוסיפים ε על האלכסון לביטחון.

C. The factorization idea

Instead of inverting R, factor it:

R = L · Lᵀ L = lower-triangular L[i][j] = 0 for j > i diagonal L[i][i] > 0 for PD matrices

Triangular systems are easy: substitute forward, then back.

R h = p L (Lᵀ h) = p (1) solve L y = p // forward substitution (2) solve Lᵀ h = y // back substitution

עברית

מפרקים R=L Lᵀ ואז פותרים שני מערכות משולשות — קדימה ואחורה. זול ויציב יותר מהיפוך מלא.

D. Formulas for L (real, column-by-column)

For i = 0..L−1, j = 0..i:

diagonal: L[i][i] = sqrt( R[i][i] − Σk=0..i−1 L[i][k]² ) below diagonal (j < i): L[i][j] = ( R[i][j] − Σk=0..j−1 L[i][k] L[j][k] ) / L[j][j]

If anything under a sqrt is ≤ 0 (or diagonal tiny), R is not PD — fail or regularize.

עברית

בונים את L שורה־שורה. אם מתחת לשורש יוצא ≤0 — המטריצה לא PD (או כמעט סינגולרית).

E. Forward and back substitution

Forward L y = p: y[i] = ( p[i] − Σj<i L[i][j] y[j] ) / L[i][i] Back Lᵀ h = y: h[i] = ( y[i] − Σj>i L[j][i] h[j] ) / L[i][i] (walk i from L−1 down to 0)

עברית

קדימה: מ־i=0 עד הסוף. אחורה: מ־i=L−1 עד 0. זה כל ה־“solve” אחרי הפירוק.

F. Regularization (DSP habit)

Near-singular R (small r[0], almost dependent rows) → Cholesky can fail or amplify noise. Common fix before factoring:

R ← R + ε · r[0] · I (ε ~ 1e−6 in LLNR WH v1)

עברית

מוסיפים ε קטן לאלכסון כדי לייצב. ב־LLNR WH זה whRegEps ≈ 1e−6.

G. Cholesky vs Levinson (so you don’t mix them)

CholeskyLevinson (general)Levinson–Durbin LPC
Solvesany SPD R h=pToeplitz R h=pYule–Walker / AR
Cost~O(L³)~O(L²)~O(p²)
LLNR WH v1yesoptional laternot the FIR feature

עברית

Cholesky = פותר SPD כללי. Levinson = פותר Toeplitz מהיר. LPC Durbin = סיפור all-pole — לא מחליף WH-FIR.

#include <cstdio>
void forward_sub(const double *L, const double *p, double *y, int n) {
    for (int i=0;i<n;i++) {
        double s=p[i];
        for (int j=0;j<i;j++) s -= L[i*n+j]*y[j];
        y[i]=s/L[i*n+i];
    }
}
int main() {
    double L[]={2,0,1,2}, p[]={4,5}, y[2];
    forward_sub(L,p,y,2);
    std::printf("y=[%.2f, %.2f]\n", y[0], y[1]);
    return 0;
}

לפי שורות, ובתוך שורה עד האלכסון — כי התלויות הן שמאלה ולמעלה.
#include <cstdio>
#include <cmath>
#include <cstring>
int chol_factor(double *L, const double *R, int n) {
    std::memset(L,0,static_cast<size_t>(n*n)*sizeof(double));
    for (int i=0;i<n;i++) {
        for (int j=0;j<=i;j++) {
            double s=R[i*n+j];
            for (int k=0;k<j;k++) s -= L[i*n+k]*L[j*n+k];
            if (i==j) {
                if (s<=0.0) return -1;
                L[i*n+i]=std::sqrt(s);
            } else {
                if (L[j*n+j]==0.0) return -1;
                L[i*n+j]=s/L[j*n+j];
            }
        }
    }
    return 0;
}
int main() {
    double R[]={4,2,0, 2,5,1, 0,1,3}, L[9];
    if (chol_factor(L,R,3)) return 1;
    for (int i=0;i<3;i++){ for(int j=0;j<3;j++) std::printf("%7.3f ",L[i*3+j]); std::putchar('\n'); }
    return 0;
}

#include <cstdio>
#include <cmath>
#include <cstring>
int chol_factor(double *L, const double *R, int n) {
    std::memset(L,0,static_cast<size_t>(n*n)*sizeof(double));
    for (int i=0;i<n;i++) for (int j=0;j<=i;j++) {
        double s=R[i*n+j];
        for (int k=0;k<j;k++) s -= L[i*n+k]*L[j*n+k];
        if (i==j) { if (s<=0) return -1; L[i*n+i]=std::sqrt(s); }
        else L[i*n+j]=s/L[j*n+j];
    }
    return 0;
}
void forward_sub(const double *L, const double *p, double *y, int n) {
    for (int i=0;i<n;i++){ double s=p[i]; for(int j=0;j<i;j++) s-=L[i*n+j]*y[j]; y[i]=s/L[i*n+i]; }
}
void back_sub(const double *L, const double *y, double *h, int n) {
    for (int i=n-1;i>=0;i--){ double s=y[i]; for(int j=i+1;j<n;j++) s-=L[j*n+i]*h[j]; h[i]=s/L[i*n+i]; }
}
int chol_solve(const double *R, const double *p, double *h, int n) {
    double L[16], y[4];
    if (chol_factor(L,R,n)) return -1;
    forward_sub(L,p,y,n); back_sub(L,y,h,n); return 0;
}
int main() {
    double R[]={4,2,2,5}, p[]={4,5}, h[2], maxr=0;
    chol_solve(R,p,h,2);
    for (int i=0;i<2;i++){ double res=-p[i]; for(int j=0;j<2;j++) res+=R[i*2+j]*h[j]; if (std::fabs(res)>maxr) maxr=std::fabs(res); }
    std::printf("h=[%.6f,%.6f] max|Rh-p|=%.3e\n", h[0],h[1], maxr);
    return 0;
}

#include <cstdio>
#include <cmath>
#include <cstring>
int chol_factor(double *L, const double *R, int n) {
    std::memset(L,0,static_cast<size_t>(n*n)*sizeof(double));
    for (int i=0;i<n;i++) for (int j=0;j<=i;j++) {
        double s=R[i*n+j]; for(int k=0;k<j;k++) s-=L[i*n+k]*L[j*n+k];
        if (i==j){ if(s<=0)return -1; L[i*n+i]=std::sqrt(s);} else L[i*n+j]=s/L[j*n+j];
    } return 0;
}
void forward_sub(const double *L, const double *p, double *y, int n) {
    for (int i=0;i<n;i++){double s=p[i]; for(int j=0;j<i;j++)s-=L[i*n+j]*y[j]; y[i]=s/L[i*n+i];}
}
void back_sub(const double *L, const double *y, double *h, int n) {
    for (int i=n-1;i>=0;i--){double s=y[i]; for(int j=i+1;j<n;j++)s-=L[j*n+i]*h[j]; h[i]=s/L[i*n+i];}
}
int main() {
    double r[3]={3,1,0.2}, R[9], L[9], p[3]={1,0.5,0.1}, y[3], h[3];
    for (int i=0;i<3;i++) for (int j=0;j<3;j++){ int d=i-j; if(d<0)d=-d; R[i*3+j]=r[d]; }
    chol_factor(L,R,3); forward_sub(L,p,y,3); back_sub(L,y,h,3);
    std::printf("h=[%.4f %.4f %.4f]\n", h[0],h[1],h[2]);
    return 0;
}

#include <cstdio>
#include <cmath>
#include <cstring>
int nr_toeplitz_solve_chol(const double *r, const double *p, double *h, int L, double regEps, double *scratch) {
    double *R = scratch;
    double *Lfac = scratch + L*L;
    double *y = scratch + 2*L*L;
    for (int i=0;i<L;i++) for (int j=0;j<L;j++){ int d=i-j; if(d<0)d=-d; R[i*L+j]=r[d]; }
    for (int i=0;i<L;i++) R[i*L+i] += regEps * r[0];
    std::memset(Lfac,0,static_cast<size_t>(L*L)*sizeof(double));
    for (int i=0;i<L;i++) for (int j=0;j<=i;j++) {
        double s=R[i*L+j];
        for (int k=0;k<j;k++) s -= Lfac[i*L+k]*Lfac[j*L+k];
        if (i==j){ if(s<=0) return -1; Lfac[i*L+i]=std::sqrt(s); }
        else Lfac[i*L+j]=s/Lfac[j*L+j];
    }
    for (int i=0;i<L;i++){ double s=p[i]; for(int j=0;j<i;j++) s-=Lfac[i*L+j]*y[j]; y[i]=s/Lfac[i*L+i]; }
    for (int i=L-1;i>=0;i--){ double s=y[i]; for(int j=i+1;j<L;j++) s-=Lfac[j*L+i]*h[j]; h[i]=s/Lfac[i*L+i]; }
    return 0;
}
int main() {
    double r[3]={3,1,0.2}, p[3]={1,0.5,0.1}, h[3], scratch[2*9+3];
    nr_toeplitz_solve_chol(r,p,h,3,1e-6,scratch);
    std::printf("h=[%.6f %.6f %.6f]\n", h[0],h[1],h[2]);
    return 0;
}
