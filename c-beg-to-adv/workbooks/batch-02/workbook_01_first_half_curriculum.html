Subject 1 — drawing shapes with prints

Medium

Context: Bench firmware prints a rectangular frame around the AFE ready message.

Requirements:

  • Print a hollow rectangle 20 characters wide and 8 lines tall using only `*` and spaces.
  • Top and bottom rows are all `*`; interior rows are `*` at column 0 and 19 only.
  • Center line 4 must contain the text `AFE READY` starting at column 5 (overwrite interior spaces).
  • Use `putchar` and/or `printf`; no loops required but loops are allowed.
#include <stdio.h>

int main(void)
{
    for (int row = 0; row < 8; ++row) {
        for (int col = 0; col < 20; ++col) {
            int edge = (row == 0 || row == 7 || col == 0 || col == 19);
            if (row == 4 && col >= 5 && col < 5 + 9) {
                char const *msg = "AFE READY";
                putchar(msg[col - 5]);
            } else if (edge) {
                putchar('*');
            } else {
                putchar(' ');
            }
        }
        putchar('\n');
    }
    return 0;
}

Hard

Context: A mini telemetry snapshot tracks running totals while samples stream in.

Requirements:

  • Start with `int frame_count = 0`, `double energy_sum = 0.0`, `double peak = 0.0`.
  • Process five samples given as literals: 0.2, -0.5, 0.5, 0.5, -0.1 (assign each to a `double sample` variable in sequence).
  • After each sample, update all three trackers.
  • Print final line: `frames=%d sum=%.4f peak=%.4f`.
  • Use a separate `double abs_sample` variable when comparing magnitude.
#include <stdio.h>

int main(void)
{
    int frame_count = 0;
    double energy_sum = 0.0;
    double peak = 0.0;
    double samples[5] = {0.2, -0.5, 0.5, 0.5, -0.1};

    for (int i = 0; i < 5; ++i) {
        double sample = samples[i];
        double abs_sample = sample < 0.0 ? -sample : sample;
        ++frame_count;
        energy_sum += sample * sample;
        if (abs_sample > peak) {
            peak = abs_sample;
        }
    }
    printf("frames=%d sum=%.4f peak=%.4f\n", frame_count, energy_sum, peak);
    return 0;
}

Context: Firmware converts a 16-bit signed ADC code to volts and must expose both integer microvolts and double volts.

Requirements:

  • Given `short raw = -8192` and `double vref = 3.3`, full scale is `32767`.
  • Compute `double volts` and `long uv` (microvolts, truncated toward zero).
  • Print `raw=%hd volts=%.6f uv=%ld`.

#include <stdio.h>

{

Hard

Context: Given left and right PCM samples, compute mono mix, peak magnitude, and headroom in dB (relative to 1.0).

Requirements:

  • `double L = 0.73`, `R = -0.41`.
  • Compute mono `m = (L + R) / 2`.
  • Compute peak `p` as the larger of `|L|`, `|R|`, `|m|`.
  • Compute `headroom_db = 20 * log10(1.0 / p)` when `p > 0`, else print `INF`.
  • Print three lines: mono, peak, headroom (use `%.4f` or `INF`).
#include <math.h>
#include <stdio.h>

int main(void)
{
    double L = 0.73;
    double R = -0.41;
    double m = (L + R) / 2.0;
    double aL = L < 0 ? -L : L;
    double aR = R < 0 ? -R : R;
    double am = m < 0 ? -m : m;
    double p = aL;
    if (aR > p) p = aR;
    if (am > p) p = am;
    printf("mono=%.4f\n", m);
    printf("peak=%.4f\n", p);
    if (p > 0.0) {
        printf("headroom_db=%.4f\n", 20.0 * log10(1.0 / p));
    } else {
        printf("headroom_db=INF\n");
    }
    return 0;
}

Context: Chain three functions: raw to volts, volts to dBFS, dBFS to LED level 0–3.

Requirements:

  • `double counts_to_volts(int c, int bits, double vref)`
  • `double volts_to_dbfs(double v, double vref)`
  • `int led_level(double dbfs)` with four bands you define in code.
  • `main` prints pipeline result for three hard-coded counts.
#include <math.h>
#include <stdio.h>

double counts_to_volts(int c, int bits, double vref)
{
    return (double)c / ((1 << bits) - 1) * vref;
}
double volts_to_dbfs(double v, double vref)
{
    if (v <= 0.0) return -120.0;
    return 20.0 * log10(v / vref);
}
int led_level(double dbfs)
{
    if (dbfs < -40.0) return 0;
    if (dbfs < -20.0) return 1;
    if (dbfs < -6.0) return 2;
    return 3;
}
int main(void)
{
    int tests[] = {100, 2000, 4000};
    for (int i = 0; i < 3; ++i) {
        double v = counts_to_volts(tests[i], 12, 3.3);
        double db = volts_to_dbfs(v, 3.3);
        printf("c=%d v=%.4f db=%.2f led=%d\n", tests[i], v, db, led_level(db));
    }
    return 0;
}

#include <stdio.h>

static int can_write(unsigned addr, unsigned val, int range_ok)
{

        printf("WRITE OK\n");
        return 1;

    return 0;
}

Context: In-range detector for stereo pairs using compound conditions.

Requirements:

  • Read `double L` and `double R`.
  • Print `IN_PHASE_OK` if both in [-1,1] AND `(L*R) >= 0`.
  • Print `LEVEL_OK` if both magnitudes <= 0.95.
  • Print `PASS` only if both conditions true; else print which failed.
  • Use `&&`, `||`, and comparisons only — no functions beyond `main`.
#include <stdio.h>

int main(void)
{
    double L = 0.0, R = 0.0;
    scanf("%lf %lf", &L, &R);
    int in_range = (L >= -1.0 && L <= 1.0 && R >= -1.0 && R <= 1.0);
    int same_sign = (L * R) >= 0.0;
    int in_phase = in_range && same_sign;
    double aL = L < 0 ? -L : L;
    double aR = R < 0 ? -R : R;
    int level_ok = (aL <= 0.95 && aR <= 0.95);
    if (in_phase) printf("IN_PHASE_OK\n"); else printf("IN_PHASE_FAIL\n");
    if (level_ok) printf("LEVEL_OK\n"); else printf("LEVEL_FAIL\n");
    printf("%s\n", (in_phase && level_ok) ? "PASS" : "FAIL");
    return 0;
}
