Subject 15 — milestone (3) project

Medium

Context: Routing console: read mode char, print delay, draw timing rail.

Requirements:

  • Read `char mode` (`L/M/H`).
  • Map to delays 2/5/9 ms with `if` chains (switch not required here if you already plan subject 16).
  • Print route line and ASCII delay rail.
  • Invalid mode → message and exit 1.
#include <stdio.h>

int main(void)
{
    char mode = '?'; int d = -1;
    printf("mode: "); scanf(" %c", &mode);
    if (mode=='L') d=2; else if (mode=='M') d=5; else if (mode=='H') d=9;
    else { printf("UNKNOWN\n"); return 1; }
    printf("delay %d ms\n", d);
    for(int i=0;i<d;++i) putchar('|'); putchar('\n');
    return 0;
}
Milestone 3 medium: integrates input, branch, and ASCII timing.
Hard

Context: Full mini-router: modes, stereo gains, and combined status word.

Requirements:

  • Read mode L/M/H and two gain integers 0–100.
  • Use `if`/`switch` acceptable; compute float gains = value/100.
  • Print delay per mode, both gains, and `STATUS` hex combining mode nibble and `(gL+gR)/2` scaled to 0–255.
  • Reject gains > 100.
#include <stdio.h>

int main(void)
{
    char mode='?'; int gL=0,gR=0,d=-1;
    scanf(" %c %d %d", &mode, &gL, &gR);
    if (gL<0||gL>100||gR<0||gR>100){ printf("RNG\n"); return 1;}
    switch(mode){ case 'L':d=2;break; case 'M':d=5;break; case 'H':d=9;break; default: printf("MODE\n"); return 1;}
    double fL=gL/100.0, fR=gR/100.0;
    unsigned status = ((unsigned)(mode & 0xF) << 8) | (unsigned)((fL+fR)*127.5);
    printf("d=%d gL=%.2f gR=%.2f status=0x%04X\n", d, fL, fR, status);
    return 0;
}
Milestone 3 hard: multi-field session with encoded status.

Subject 16 — switch case

Medium

Context: Map filter profile ID 1–4 to names and tap counts.

Requirements:

  • Read `int profile`.
  • Switch: 1=COMB2, 2=COMB4, 3=ALLPASS, 4=FLAT.
  • Default: `UNKNOWN`.
  • Print `profile %d => %s taps %d` with taps 2,4,6,1 respectively.
#include <stdio.h>

int main(void)
{
    int profile = 0; scanf("%d", &profile);
    char const *name = "UNKNOWN"; int taps = 0;
    switch (profile) {
    case 1: name = "COMB2"; taps = 2; break;
    case 2: name = "COMB4"; taps = 4; break;
    case 3: name = "ALLPASS"; taps = 6; break;
    case 4: name = "FLAT"; taps = 1; break;
  default: break;
    }
    printf("profile %d => %s taps %d\n", profile, name, taps);
    return 0;
}
Straightforward int switch with default.
Hard

Context: Fall-through style grouping: band letter sets FFT size tier.

Requirements:

  • Read `char band` (`A`–`D`).
  • Switch with grouped cases: A/B → 256, C → 512, D → 1024, else 0.
  • Print `band %c fft %d` and `TIER n` where tier 1/2/3 matches size.
  • Use at least one intentional fall-through group.
#include <stdio.h>

int main(void)
{
    char band = '?'; scanf(" %c", &band);
    int fft = 0, tier = 0;
    switch (band) {
    case 'A':
    case 'B':
        fft = 256; tier = 1; break;
    case 'C':
        fft = 512; tier = 2; break;
    case 'D':
        fft = 1024; tier = 3; break;
    default:
        fft = 0; tier = 0; break;
    }
    printf("band %c fft %d\nTIER %d\n", band, fft, tier);
    return 0;
}

Subject 17 — structs

Medium

Context: Define `struct Sample` with frame id and value; print and compare.

Requirements:

  • Fields: `int id`, `double value`.
  • Initialize two structs; print both.
  • Print `EQ` or `NE` if both fields match.
#include <stdio.h>

struct Sample { int id; double value; };

int main(void)
{
    struct Sample a = {1, 0.125};
    struct Sample b = {1, 0.125};
    printf("%d %.4f\n", a.id, a.value);
    printf("%d %.4f\n", b.id, b.value);
    printf("%s\n", (a.id==b.id && a.value==b.value) ? "EQ" : "NE");

Basic struct define, init, field access.
Hard

Context: Stereo frame struct with tag; compute peak and print CSV line.

Requirements:

  • `struct Frame { int seq; double l; double r; char tag[10]; }`
  • Function `double frame_peak(struct Frame f)`
  • Function `void print_csv(struct Frame f)` → `seq,l,r,tag`
  • Process array of 3 frames in `main`.
#include <stdio.h>

struct Frame { int seq; double l; double r; char tag[10]; };

double frame_peak(struct Frame f)
{
    double aL = f.l < 0 ? -f.l : f.l;
    double aR = f.r < 0 ? -f.r : f.r;
    return aL > aR ? aL : aR;
}

void print_csv(struct Frame f)
{
    printf("%d,%.5f,%.5f,%s\n", f.seq, f.l, f.r, f.tag);
}

int main(void)
{
    struct Frame buf[3] = {
        {1, 0.2, -0.1, "A"},
        {2, 0.9, 0.8, "B"},
        {3, -0.3, 0.1, "C"}
    };
    for (int i = 0; i < 3; ++i) {
        print_csv(buf[i]);
        printf("peak=%.4f\n", frame_peak(buf[i]));
    }
    return 0;
}

Subject 18 — while loops

Medium

Context: Count samples until sentinel `-1` is read.

Requirements:

  • Read ints in a `while` loop; stop before processing sentinel.
  • Track count and running sum.
  • Print `n=%d sum=%d`.
#include <stdio.h>

int main(void)
{
    int stream[] = {3,5,2,8,-1};
    int i = 0, n = 0, sum = 0, x = 0;
    x = stream[i++];
    while (x != -1) {
        ++n; sum += x;
        x = stream[i++];
    }
    printf("n=%d sum=%d\n", n, sum);
    return 0;
}
While with sentinel termination.
Hard

Context: Read doubles until `0.0` sentinel; track clips above 0.98 and longest streak of positives.

Requirements:

  • Use `while`; update clip count and streak logic.
  • Print `count clips=%d max_streak=%d`.
#include <stdio.h>

int main(void)
{
    double s[] = {0.2, 0.99, 0.4, 1.0, 0.1, -0.2, 0.3, 0.0};
    int i = 0, clips = 0, streak = 0, max_streak = 0;
    double x = s[i++];
    while (x != 0.0) {
        if (x > 0.98) ++clips;
        if (x > 0.0) { ++streak; if (streak > max_streak) max_streak = streak; }
        else streak = 0;
        x = s[i++];
    }
    printf("clips=%d max_streak=%d\n", clips, max_streak);
    return 0;
}

Subject 19 — milestone (4) project

Medium

Context: Session logger: capture N frames in arrays, dump table to stdout.

Requirements:

  • Read `n` (1–8) and `n` pairs of `(id, power)`.
  • Store in parallel arrays.
  • Print markdown-style table with header row.
  • Print id of max power.
#include <stdio.h>

int main(void)
{
    int n=0; scanf("%d",&n);
    int id[8]={0}; double p[8]={0};
    for(int i=0;i<n;++i) scanf("%d %lf",&id[i],&p[i]);
    printf("| id | power |\n");
    int best=0;
    for(int i=0;i<n;++i){
        printf("| %2d | %7.4f |\n", id[i], p[i]);
        if(p[i]>p[best]) best=i;
    }
    printf("max_id=%d\n", id[best]);
    return 0;
}
Milestone 4 medium: capture + report without files.
Hard

Context: Capture session then write `session.txt` and read back row count.

Requirements:

  • Same capture as medium.
  • Write one line per frame: `id,power`.
  • Reopen and count lines with while+fgets.
  • Print `WROTE n READ m` and `MATCH` if equal.
#include <stdio.h>

int main(void)
{
    int n=0; scanf("%d",&n);
    int id[8]={0}; double p[8]={0};
    for(int i=0;i<n;++i) scanf("%d %lf",&id[i],&p[i]);
    FILE *fw = fopen("session.txt","w");
    for(int i=0;i<n;++i) fprintf(fw,"%d,%.6f\n",id[i],p[i]);
    fclose(fw);
    int m=0; char line[64];
    FILE *fr = fopen("session.txt","r");
    while(fgets(line,sizeof line,fr)) ++m;
    fclose(fr);
    printf("WROTE %d READ %d\n", n, m);
    printf("%s\n", n==m?"MATCH":"MISMATCH");
    return 0;
}

Context: Pointer walk: compute mean of `double` array using `begin` and `end` pointers only.

Requirements:

  • No subscript in mean function.
#include <stdio.h>

double mean_ptr(double const *begin, double const *end)
{
    int n = 0;
    double sum = 0.0;
    for (double const *p = begin; p < end; ++p) {
        sum += *p;
        ++n;
    }
    return n ? sum / n : 0.0;
}

int main(void)
{
    double a[5] = {1,2,3,4,5};
    printf("%.2f\n", mean_ptr(a, a+5));
    return 0;
}
