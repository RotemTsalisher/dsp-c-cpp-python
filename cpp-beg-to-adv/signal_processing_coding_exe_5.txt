#include <iostream>
#include <type_traits>
#include <concepts>

template <typename T, size_t N>
requires std::is_arithmetic_v<T>
T dot_in_place(const T (&x1)[N], const T (&x2)[N]){
    
    T res = 0;
    for (std::size_t i = 0; i< N ; res += (x1[i]*x2[i++]));
    return res;
};

int main()
{
    // Test 1: Integer array (Size deduced as 3)
    int a[] = {1, 2, 3};
    int b[] = {4, 5, 6};
    int res_int = dot_in_place(a, b); // 1*4 + 2*5 + 3*6 = 32
    std::cout << "Integer Dot Product: " << res_int << std::endl;

    // Test 2: Floating point (Size deduced as 4)
    // Useful for your 1/3 octave band signal processing
    double x[] = {0.5, 1.0, 2.0, 3.0};
    double y[] = {1.5, 2.0, 0.5, 1.0};
    double res_db = dot_in_place(x, y); // 0.75 + 2.0 + 1.0 + 3.0 = 6.75
    std::cout << "Double Dot Product:  " << res_db << std::endl;

    // Test 3: Const-correctness check
    const float f1[] = {1.1f, 2.2f};
    const float f2[] = {3.3f, 4.4f};
    std::cout << "Const Float Product: " << dot_in_place(f1, f2) << std::endl;

    return 0;
}
