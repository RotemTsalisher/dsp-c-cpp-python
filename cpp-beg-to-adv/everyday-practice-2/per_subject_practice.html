03functions templates

Template template<typename T> T lerp(T a, T b, double t) with t clamped to [0,1] using comparisons only (no algorithms header).

template <typename T>
T lerp(T const a, T const b, double t)
{
    if (t < 0.0) {
        t = 0.0;
    }
    if (t > 1.0) {
        t = 1.0;
    }
    return static_cast<T>(a + static_cast<T>((b - a) * t));
}

05lambda functions

Lambda capturing double alpha by value: given previous smoothed value yPrev and new input x, return one-pole low-pass yPrev + alpha * (x - yPrev) (IIR coefficient drill).

double smoothStep(double const alpha, double const yPrev, double const x)
{
    auto const step = [alpha](double const y0, double const xi) {
        return y0 + alpha * (xi - y0);
    };
    return step(yPrev, x);
}
06requires

template<typename T> constexpr T db20(T linear) with requires that T is floating-point (std::is_floating_point_v<T>). Use 10*log10 for positive linear; return 0 otherwise.

#include <cmath>
#include <type_traits>

template <typename T>
requires std::is_floating_point_v<T>
constexpr T db20(T const linear)
{
    if (linear <= T{0}) {
        return T{0};
    }
    return static_cast<T>(10.0 * std::log10(static_cast<double>(linear)));
}

12threads (<thread> basics)

std::thread sums four fixed double values into double acc via lambda (by reference), then join() — toy parallel PSD bin accumulation.

#include <thread>

int main()
{
    double acc = 0.0;
    double const b0 = 0.1;
    double const b1 = 0.2;
    double const b2 = 0.3;
    double const b3 = 0.4;
    std::thread worker([&acc, b0, b1, b2, b3]() { acc = b0 + b1 + b2 + b3; });
    worker.join();
    return 0;
}
