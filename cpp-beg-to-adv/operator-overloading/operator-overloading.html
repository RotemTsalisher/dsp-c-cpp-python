C++ Operator Overloading Worksheet

Starter Class

class Vec2 {
public:
    double x, y;

    Vec2(double x = 0.0, double y = 0.0) : x(x), y(y) {}

    void print() const {
        std::cout << "(" << x << ", " << y << ")\n";
    }
};

Part 1 — Member Operators

Exercise 1.1 — Vec2 + Vec2

Vec2 operator+(const Vec2& other) const;
Vec2 operator+(const Vec2& other) const {
    return Vec2(x + other.x, y + other.y);
}

Exercise 1.2 — +=

Vec2& operator+=(const Vec2& other);
Vec2& operator+=(const Vec2& other) {
    x += other.x;
    y += other.y;
    return *this;
}

Exercise 1.3 — Unary -

Vec2 operator-() const {
    return Vec2(-x, -y);
}

Part 2 — Left vs Right Operands

Exercise 2.1 — Support both sides

Vec2 operator+(const Vec2& v, double s);
Vec2 operator+(double s, const Vec2& v);

Vec2 operator+(const Vec2& v, double s) {
    return Vec2(v.x + s, v.y + s);
}

Vec2 operator+(double s, const Vec2& v) {
    return Vec2(v.x + s, v.y + s);
}

Part 3 — Scalar Multiplication

Vec2 operator*(double s) const {
    return Vec2(x * s, y * s);
}

Vec2 operator*(double s, const Vec2& v) {
    return Vec2(v.x * s, v.y * s);
}

Vec2& operator*=(double s) {
    x *= s;
    y *= s;
    return *this;
}

Part 4 — Comparison

bool operator==(const Vec2& other) const {
    return x == other.x && y == other.y;
}

bool operator!=(const Vec2& other) const {
    return !(*this == other);
}

Part 5 — Stream Operator

std::ostream& operator<<(std::ostream& os, const Vec2& v) {
    os << "(" << v.x << "," << v.y << ")";
    return os;
}

Part 6 — Index Operator

double& operator[](int index) {
    return (index == 0) ? x : y;
}

const double& operator[](int index) const {
    return (index == 0) ? x : y;
}

Part 7 — Dot Product

double operator*(const Vec2& a, const Vec2& b) {
    return a.x * b.x + a.y * b.y;
}

Part 8 — Prefix/Postfix ++

Vec2& operator++() {
    ++x; ++y;
    return *this;
}

Vec2 operator++(int) {
    Vec2 temp = *this;
    ++(*this);
    return temp;
}

Full Test Main

int main() {
    Vec2 a(1,2), b(3,4);

    (a + b).print();
    (a + 3).print();
    (3 + a).print();

    (a * 2).print();
    (2 * a).print();

    std::cout << (a == b) << std::endl;
    std::cout << a << std::endl;

    a[0] = 10;
    a.print();

    std::cout << (a * b) << std::endl;

    return 0;
}
