#include <iostream>

class Base {
    private:
        int private_int;
    protected:
        int protected_int;
    public:
        int public_int;
        
        Base() : private_int(-1), protected_int(-1), public_int(-1) {};
        Base(int a, int b, int c) : private_int(a), protected_int(b), public_int(c) {};
        
        void print_info() {
            std::cout << " --- Base:" << std::endl;
            std::cout << "private: " << private_int << std::endl;
            std::cout << "protected: " << protected_int << std::endl;
            std::cout << "public: " << public_int << std::endl << std::endl;
        };
        
        friend std::ostream& operator<<(std::ostream& os, const Base& b) {
            os << " --- Base:" << std::endl;
            os << "private: " << b.private_int << std::endl;
            os << "protected: " << b.protected_int << std::endl;
            os << "public: " << b.public_int << std::endl << std::endl;
            return os;
        };
        
        int get_private_info() const {return private_int;};
};

class Derived : public Base {
    
    private:
        int private_int;
    protected:
        int protected_int;
    public:
        int public_int;
        
    Derived() : Base(), private_int(0), protected_int(0), public_int(0) {};
    Derived(int a, int b, int c) : Base(), private_int(a), protected_int(b), public_int(c) {};
    Derived(int a, int b, int c, int d, int e, int f) : Base(a,b,c), private_int(d), protected_int(e), public_int(f) {};
    
    int get_base_private_info() const {return Base::get_private_info(); };
    
    void print_info() {
            Base::print_info();
            std::cout << " --- Derived:" << std::endl;
            std::cout << "private: " << private_int << std::endl;
            std::cout << "protected: " << protected_int << std::endl;
            std::cout << "public: " << public_int << std::endl << std::endl;
        };
    
    friend std::ostream& operator<<(std::ostream& os, const Derived& d) {
            
            os << static_cast<const Base&>(d);
            os << " --- Derived:" << std::endl;
            os << "private: " << d.private_int << std::endl;
            os << "protected: " << d.protected_int << std::endl;
            os << "public: " << d.public_int << std::endl << std::endl;
            return os;
    };
};

int main()
{
    /*
    Base b0;
    Base b1(1,2,3);
    
    std::cout << b0;
    std::cout << b1;
    */
    
    Derived d0;
    Derived d1(10,20,30);
    Derived d2(1,2,3,4,5,6);
    
    d0.print_info();
    d1.print_info();
    d2.print_info();
    
    return 0;
}
