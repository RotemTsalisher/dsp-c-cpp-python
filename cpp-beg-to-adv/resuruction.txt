#include <print>
#include <iostream>

class Base {
    private:
        int private_int;
    protected:
        int protected_int;
    public:
        int public_int;
    
        Base() : private_int(-1), protected_int(-1), public_int(-1) {};
        Base(int a, int b, int c) : private_int(a), protected_int(b), public_int(c) {};
        
        friend std::ostream& operator<<(std::ostream& os, const Base& b) {
            os << "Base: " << std::endl;
            os << "private_int   = " << b.private_int   << std::endl;
            os << "protected_int = " << b.protected_int << std::endl;
            os << "public_int    = " << b.public_int    << std::endl;
            return os;
        };
};

class Derived : private Base {
    private:
        int private_int;
    protected:
       // int protected_int;
    public:
        int public_int;
        using Base::protected_int;
        Derived() : Base(), private_int(0), protected_int(0), public_int(0) {};
        Derived(int a, int b, int c) : Base(), private_int(a), protected_int(b), public_int(c) {};
        Derived(int a, int b, int c, int d, int e, int f) : Base(a,b,c), private_int(d), protected_int(e), public_int(f) {};
        
        
};
