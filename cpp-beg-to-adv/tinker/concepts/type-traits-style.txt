template <typename T>
concept demand_ = std::integral<T> || std::floating_point<T>;
void print_(demand_ auto a){
    
    std::cout << "== EITHER INTEGRAL OR FLOATING POINT ==" << std::endl;
    std::cout << "Value: " << a << std::endl;
}

template <typename T>

void print_(demand_str auto a){
    std::cout << "== STR ONLY ==" << std::endl;

};
