=== requires + type specialization ===

#include <iostream>
#include <concepts>
#include <string>

template <typename T>
requires (std::integral<T> || std::floating_point<T>)
void printVal(T a){
    std::cout << "Value: " << a << std::endl;
};

template <typename T>
requires std::same_as<T, std::string>
void printVal(T a){
    std::cout << a << std::endl;
}

int main()
{
   
    unsigned int a = 4;
    unsigned char a_ = 2;
    int b = 1;
    
    float f = 0.01f;
    double f_ = 0.02;
    
    printVal(a);
    printVal(static_cast<char>(a_ + '0'));
    
    printVal(b);
    
    printVal(f);
    printVal(f_);
    
    std::string messege = "Hello World!";
    printVal(messege);
    return 0;
}
