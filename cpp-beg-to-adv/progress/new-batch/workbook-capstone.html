Capstone workbook — exercises combining the full course

Scope: Same rule as the batches: use only mechanisms that appear on your course-subjects.txt list. No standard containers, smart pointers, mutexes, algorithms, std::function, std::invocable, std::string, etc. Verify with return / static_assert instead of I/O.

1 Template + lambda + raw pointers

Task: template<typename Pred> int countIf(int const* first, int const* last, Pred p) counts elements in [first, last) for which p(*it) is true. Test with C array int a[] = {1,2,3,4}; and a lambda counting even values; expect 2.

template <typename Pred>
int countIf(int const* first, int const* last, Pred p) {
    int n = 0;
    for (; first != last; ++first)
        if (p(*first)) ++n;
    return n;
}

int main() {
    int a[] = {1, 2, 3, 4};
    int n = countIf(a, a + 4, [](int x) { return x % 2 == 0; });
    return n == 2 ? 0 : 1;
}

3 Encapsulation + pointers (list: managing objects through pointers)

Task: Class Wallet with private int balance_, ctor, int balance() const, void add(int x). In main, Wallet* w = new Wallet(10);, add, read balance, delete w;. No smart pointers.

class Wallet {
    int balance_{};
public:
    explicit Wallet(int b = 0) : balance_(b) {}
    int balance() const { return balance_; }
    void add(int x) { balance_ += x; }
};

int main() {
    Wallet* w = new Wallet(10);
    w->add(5);
    int b = w->balance();
    delete w;
    return b == 15 ? 0 : 1;
}

4 Virtual + destructor + inheritance

Task: Base Source with virtual int next() = 0; and virtual ~Source() = default;. ConstantSource returns a fixed int. Delete through Source* after new ConstantSource(7) and check value before delete.

struct Source {
    virtual int next() = 0;
    virtual ~Source() = default;
};

struct ConstantSource : Source {
    int v;
    explicit ConstantSource(int x) : v(x) {}
    int next() override { return v; }
};

int main() {
    Source* p = new ConstantSource(7);
    int x = p->next();
    delete p;
    return x == 7 ? 0 : 1;
}

7 Threads (basics only)

Task: Same pattern as Batch 2: two threads, each writes into its own int through a distinct int* (e.g. compute a small sum). Join both; no mutex. Confirms std::thread + join only.

#include <thread>

void triple(int* out) { *out = 3 * 4; }

int main() {
    int a = 0, b = 0;
    std::thread t1(triple, &a);
    std::thread t2(triple, &b);
    t1.join();
    t2.join();
    return (a == 12 && b == 12) ? 0 : 1;
}

8 Fold + function pointer table

Task: template<typename... Ts> constexpr int typeCount() as sizeof...(Ts). C array of two int(*)(int,int) for add/mul; dispatch by index. static_assert(typeCount<int,char,double>() == 3);

template <typename... Ts>
constexpr int typeCount() {
    return sizeof...(Ts);
}

static_assert(typeCount<int, char, double>() == 3);

int add(int a, int b) { return a + b; }
int mul(int a, int b) { return a * b; }

using BinOp = int (*)(int, int);

int dispatch(BinOp const ops[2], int idx, int x, int y) {
    return ops[idx](x, y);
}

int main() {
    BinOp ops[2] = {add, mul};
    return dispatch(ops, 0, 2, 3) == 5 && dispatch(ops, 1, 2, 3) == 6 ? 0 : 1;
}

9 Polymorphism + const + raw pointers (fixed small chain)

Task: Abstract struct Stage { virtual double run(double x) const = 0; virtual ~Stage() = default; }; GainStage returns g*x. ClipStage clamps to [-1,1] with plain comparisons (no std::clamp). Build a chain of exactly two stages using Stage const* chain[2] (or two locals and pointers). Pass 0.5 through both; delete any new’d objects. Const-correct run.

struct Stage {
    virtual double run(double x) const = 0;
    virtual ~Stage() = default;
};

struct GainStage : Stage {
    double g;
    explicit GainStage(double gain) : g(gain) {}
    double run(double x) const override { return g * x; }
};

struct ClipStage : Stage {
    double run(double x) const override {
        if (x < -1.0) return -1.0;
        if (x > 1.0) return 1.0;
        return x;
    }
};

double runChain(Stage const* const stages[], int n, double x) {
    double y = x;
    for (int i = 0; i < n; ++i)
        y = stages[i]->run(y);
    return y;
}

int main() {
    GainStage* g = new GainStage(2.0);
    ClipStage* c = new ClipStage();
    Stage const* chain[2] = {g, c};
    double out = runChain(chain, 2, 0.5); // 0.5*2 = 1.0, clip stays 1.0
    delete g;
    delete c;
    return out == 1.0 ? 0 : 1;
}
