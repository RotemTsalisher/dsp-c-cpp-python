Exercise 7 focuses on hiding vs override and is the hardest in this batch.

Exercise 1protected

Task: Class Base with protected: int n_; and public ctor setting n_. Class Derived : public Base with method int bump() that increments n_ and returns it. main cannot access n_ on a Derived from outside the class (verify).

class Base {
protected:
    int n_;
public:
    explicit Base(int n = 0) : n_(n) {}
};

class Derived : public Base {
public:
    using Base::Base;
    int bump() { return ++n_; }
};

Exercise 2Base access specifiers

Task: Write three one-line comments comparing struct D1 : public B, struct D2 : protected B, struct D3 : private B regarding how B’s public members appear to users of D1/D2/D3. No code required beyond a minimal empty struct B {}; if you want to compile.

struct B { public: int x; };

// public inheritance: B's public stays public to users of D1.
struct D1 : public B {};

// protected inheritance: B's public becomes protected in D2 (outside world cannot use x).
struct D2 : protected B {};

// private inheritance: B's public becomes private in D3 (implementation detail).
struct D3 : private B {};

Exercise 3friend

Task: class Engine { friend class Mechanic; private: int secret_; public: explicit Engine(int s):secret_(s){} }; and class Mechanic { public: static int peek(Engine const& e) { return e.secret_; } };. Only Mechanic can read secret_. Demonstrate in main.

class Engine {
    friend class Mechanic;
    int secret_;
public:
    explicit Engine(int s) : secret_(s) {}
};

class Mechanic {
public:
    static int peek(Engine const& e) { return e.secret_; }
};

// Mechanic::peek(Engine(3)) == 3 ; direct e.secret_ in main is ill-formed

Task: struct A { int x; }; struct B : A { int y; }; Write copy constructor for B that copies both x and y from another B (use initializer list : A(other), y(other.y)).

struct A {
    int x{};
};

struct B : A {

    B() = default;

Exercise 6Virtual

Task: Abstract interface struct IFilter { virtual double process(double x) const = 0; virtual ~IFilter() = default; }; Concrete Gain : IFilter with member double g_ and process returning g_*x. Call through IFilter const*.

struct IFilter {
    virtual double process(double x) const = 0;
    virtual ~IFilter() = default;
};

struct Gain : IFilter {
    double g_;
    explicit Gain(double g) : g_(g) {}
    double process(double x) const override { return g_ * x; }
};

Exercise 7Overload / override / hiding

Task: (1) Show that a derived class member void f(int) hides all overloads f(double) from base unless you import with using Base::f;. (2) Show correct override of virtual void g(); in derived. Write minimal structs and comments explaining the three keywords overload (same name, different params), override (virtual replacement), hiding (name in derived shadows base).

struct Base {
    void f(double) {}
    virtual void g() {}
};

struct DerivedHide : Base {
    void f(int) {} // hides Base::f(double) — d.f(1.0) won't resolve to Base::f
};

struct DerivedUsing : Base {
    using Base::f;
    void f(int) {} // both overloads visible where using applies
};

struct DerivedOverride : Base {
    void g() override {} // replaces virtual g(); use override keyword
};

// Overload: two functions same name, different signatures.
// Override: derived virtual function replaces base virtual with same signature.
// Hiding: derived name f shadows all Base::f unless brought in with using.
