Exercise 1this

Task: Class Node: Node& setValue(int v) sets value_ and returns *this; private int value_; int value() const. Chain n.setValue(1).setValue(2) and check value() == 2 via return.

class Node {
    int value_{};
public:
    Node& setValue(int v) {
        value_ = v;
        return *this;
    }
    int value() const { return value_; }
};

int main() {
    Node n;
    n.setValue(1).setValue(2);
    return n.value() == 2 ? 0 : 1;
}

float normSq(Vec2 v) {
    return v.x * v.x + v.y * v.y;
}

Exercise 4Inheritance

Task: struct Base { virtual int idCode() const; }; returns 1. struct Derived : Base { int idCode() const override; } returns 2. Through Base const& ref = d;, ref.idCode() == 2. No I/O—use return.

struct Base {
    virtual int idCode() const { return 1; }
};

struct Derived : Base {
    int idCode() const override { return 2; }
};

int main() {
    Derived d;
    Base const& ref = d;
    return ref.idCode() == 2 ? 0 : 1;
}

Exercise 5const correctness

Task: Class Book with private int pages_. int pages() const and non-const void setPages(int p). On a const Book, only pages() may be called. Demonstrate with compile-time reasoning in comments; in code, use non-const object in main returning pages() after setPages.

class Book {
    int pages_{0};
public:
    int pages() const { return pages_; }
    void setPages(int p) { pages_ = p; }
};

// const Book b; b.setPages(3); // ill-formed: setPages is not const

int main() {
    Book b;
    b.setPages(120);
    return b.pages() == 120 ? 0 : 1;
}

Exercise 6const and reference members

Task: RefHolder with const int id_ and int& ref_, ctor initializer list only. int x=5; RefHolder h(1,x); x=9; then h.refValue()==9.

class RefHolder {
    const int id_;
    int& ref_;
public:
    RefHolder(int id, int& r) : id_(id), ref_(r) {}
    int id() const { return id_; }
    int refValue() const { return ref_; }
};

int main() {
    int x = 5;
    RefHolder h(1, x);
    x = 9;
    return h.refValue() == 9 ? 0 : 1;
}

Exercise 7Chaining

Task: Builder with withA, withB, sum as in the original batch; verify Builder{}.withA(3).withB(4).sum() == 7 via return.

class Builder {
    int a_{0};
    int b_{0};
public:
    Builder& withA(int x) { a_ = x; return *this; }
    Builder& withB(int y) { b_ = y; return *this; }
    int sum() const { return a_ + b_; }
};

int main() {
    int v = Builder{}.withA(3).withB(4).sum();
    return v == 7 ? 0 : 1;
}
