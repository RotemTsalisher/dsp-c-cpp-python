🎯 C++ Exercise: Abstract Codec

Goal

Build a small polymorphic system for encoding audio frames using an abstract base class.

Requirements

  • Create an abstract class Codec
  • It must contain:
    • A pure virtual function:
      virtual void encodeFrame(double const* input, int numSamples) = 0;
    • A protected default constructor
    • A virtual destructor

Your Tasks

  1. Implement a derived class PCMCodec
    • It should "encode" by printing the samples as-is
  2. Implement another derived class QuantizedCodec
    • It should convert samples into integers (simulate quantization)
  3. In main():
    • Create a sample frame (array of doubles)
    • Use a Codec* pointer
    • Call encodeFrame on both implementations

Constraints

  • You must NOT instantiate Codec directly
  • You must use polymorphism (base pointer)

// ===== Solution =====

#include <iostream>

class Codec {
public:
    virtual ~Codec() = default;
    virtual void encodeFrame(double const* input, int numSamples) = 0;

protected:
    Codec() = default;
};

class PCMCodec : public Codec {
public:
    void encodeFrame(double const* input, int numSamples) override {
        std::cout << "PCM: ";
        for (int i = 0; i < numSamples; ++i)
            std::cout << input[i] << " ";
        std::cout << std::endl;
    }
};

class QuantizedCodec : public Codec {
public:
    void encodeFrame(double const* input, int numSamples) override {
        std::cout << "Quantized: ";
        for (int i = 0; i < numSamples; ++i)
            std::cout << static_cast<int>(input[i] * 10) << " ";
        std::cout << std::endl;
    }
};

int main() {
    double frame[] = {0.1, 0.5, -0.3, 0.8};

    Codec* c1 = new PCMCodec();
    Codec* c2 = new QuantizedCodec();

    c1->encodeFrame(frame, 4);
    c2->encodeFrame(frame, 4);

    delete c1;
    delete c2;
}
