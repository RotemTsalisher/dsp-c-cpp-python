#include <iostream>
#include <string>
#include <type_traits>

template <typename S>
//requires (std::is_integral_v<S> || std::is_array_v<S>)
void smartPrint(S& s){
    constexpr size_t size = std::extent_v<S>;
    static_assert(size >= 2, "Less than 2 elements!");
    
    auto sum = s[0] - s[0];
    for(size_t i = 0; i< size ; sum += s[i], i++){
        std::cout << "[" << sum << "] ";
    }
    std::cout << std::endl;
    std::cout << "sum = " << sum << std::endl;
};

int main()
{
    double arr[] = {0.2, 1.1, 2.7, 3.6, 3.2, 2.6, 1.8};
    double val = 2.2;
    smartPrint(arr);
    //smartPrint(val);
    return 0;
}
