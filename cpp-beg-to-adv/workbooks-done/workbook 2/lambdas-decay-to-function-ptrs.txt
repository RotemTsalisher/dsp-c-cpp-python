#include <iostream>
#include <string>

int main()
{
    unsigned int userSelection = 1;
    int a = 1, b = 2;
    
    auto add   = [](int a, int b) -> int {return a + b;};
    auto mult  = [](int a, int b) -> int {return a * b;};
    
    int(*op)(int,int);
    if(0 == userSelection){
        op = add;
    }
    else{
        op = mult;
    }
    
    std::cout << "op returns: " << op(a,b) << std::endl;
    return 0;
}
