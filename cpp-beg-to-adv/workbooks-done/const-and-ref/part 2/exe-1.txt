#include <iostream>

class Modifier {
    private:
        int& val;
    
    public:
        Modifier(int& v) : val(v) {};
        Modifier(const Modifier& m) : val(m.val) {};
        
        Modifier& increment_value() {val++ ; return *this;};
        Modifier& print_value() {std::cout << "Value: " << val << std::endl; return *this;};
        int get_value() {return val; };
};

int main()
{
    int a = 3;
    Modifier m(a);
    
    m.increment_value().print_value();
    std::cout << "a = " << a << std::endl;
    return 0;
}
