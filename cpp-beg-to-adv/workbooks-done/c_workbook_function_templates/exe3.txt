#include <iostream>

auto square_(auto x){
    decltype(auto) res = x*x;
    return res;
}

void swap(int& a, int& b){
    int tmp = a;
    a = b;
    b = tmp;
}

void print(int a) {
    std::cout << "print(int): n = " << a << std::endl;
}
void print(double a) {
    std::cout << "print(double): n = " << a << std::endl;
}
int main()
{
    print(1);
    print(1.2);
    print(2.5f);
    return 0;
}
