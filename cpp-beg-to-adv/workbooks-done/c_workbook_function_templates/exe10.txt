#include <iostream>

int return_int(void){
    return 1;
};

auto call_func(auto (*f)(void)){
    return f();
};

template <typename... Args>
auto sum_args(Args... args){
    return (args + ...);
}
int main()
{
    int (*foo)(void) = &return_int;
    
    std::cout << "return_int returned: " << call_func(foo) << std::endl;
    std::cout << "sum_args returned: " << sum_args(1,2,3.2,-4,5) << std::endl;
    return 0;
}
