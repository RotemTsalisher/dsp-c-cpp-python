Section 1 – Friend ostream Printing

Exercise 1 – Printing a Point

Create a class Point with private members:

int x;
int y;

Implement a friend operator<< so this works:

Point p(3,7);
std::cout << p << std::endl;
#include <iostream>

class Point{
private:
    int x;
    int y;

public:
    Point(int x_, int y_) : x(x_), y(y_) {}

    friend std::ostream& operator<<(std::ostream& os, const Point& p){
        os << "Point(" << p.x << ", " << p.y << ")";
        return os;
    }
};

Exercise 2 – Rectangle Printer

Create a class Rectangle with:

int width;
int height;

Add a function:

int area() const;

Print format:

Rectangle[width=10, height=5, area=50]
#include <iostream>

class Rectangle{
private:
    int width;
    int height;

public:
    Rectangle(int w,int h):width(w),height(h){}

    int area() const { return width*height; }

    friend std::ostream& operator<<(std::ostream& os,const Rectangle& r){
        os << "Rectangle[width=" << r.width
           << ", height=" << r.height
           << ", area=" << r.area() << "]";
        return os;
    }
};

Exercise 3 – Nested Objects

Create classes Point and Line.

Line contains:

Point start;
Point end;

Output format:

Line[Point(1,2) -> Point(5,8)]
#include <iostream>

class Point{
private:
    int x,y;

public:
    Point(int x_,int y_):x(x_),y(y_){}

    friend std::ostream& operator<<(std::ostream& os,const Point& p){
        os << "Point(" << p.x << "," << p.y << ")";
        return os;
    }
};

class Line{
private:
    Point start;
    Point end;

public:
    Line(Point s,Point e):start(s),end(e){}

    friend std::ostream& operator<<(std::ostream& os,const Line& l){
        os << "Line[" << l.start << " -> " << l.end << "]";
        return os;
    }
};

Exercise 4 – Vector3D

Create class Vector3D:

double x;
double y;
double z;

Add:

double magnitude() const;

Example output:

Vector3D(x=1, y=2, z=3, |v|=3.74)
#include <iostream>
#include <cmath>

class Vector3D{
private:
    double x,y,z;

public:
    Vector3D(double x_,double y_,double z_):x(x_),y(y_),z(z_){}

    double magnitude() const{
        return std::sqrt(x*x + y*y + z*z);
    }

    friend std::ostream& operator<<(std::ostream& os,const Vector3D& v){
        os << "Vector3D(x=" << v.x
           << ", y=" << v.y
           << ", z=" << v.z
           << ", |v|=" << v.magnitude() << ")";
        return os;
    }
};
