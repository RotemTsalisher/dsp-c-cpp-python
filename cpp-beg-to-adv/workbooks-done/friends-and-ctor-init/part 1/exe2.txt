#include <iostream>

class Rectangle {
    
    private:
        int height, width;
    
    public:
        Rectangle() = default;
        Rectangle(int h, int w) : height(h), width(w) {};
        Rectangle(const Rectangle& r) : height(r.height), width(r.width) {};
        
        Rectangle* set_height(int h) {height = h; return this;};
        Rectangle* set_width(int w) {width = w; return this;};
        
        int get_height() const {return height;};
        int get_width() const {return width;};
        
        int area() const {return height*width;};
        
        friend std::ostream& operator<<(std::ostream& os, const Rectangle& r) {
            std::cout << "Rectangle[width = " << r.width << ", height = " << r.height << ", area = " << r.area() << "]";
            return os;
            
        };
};

int main(){
    
    Rectangle r(2,4);
    std::cout << r << std::endl;
}
