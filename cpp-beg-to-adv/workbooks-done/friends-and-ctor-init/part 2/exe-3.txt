#include <iostream>

class Point {
    private:
        int x,y;
    public:
        Point(int x_, int y_) : x(x_), y(y_) {};
        int get_x() {return x;};
        int get_y() {return y;};
};

class Circle {
    private:
        Point m;
        double radius;
    
    public:
        Circle(int a, int b, double r_) : m(a, b), radius(r_) {};
        int get_x() {return m.get_x();};
        int get_y() {return m.get_y();};
        double get_radius() {return radius;};
};


int main(){
    Circle c(2,4, 1);
    std::cout << "c:" << std::endl;
    std::cout << "x = " << c.get_x() << ", y = " << c.get_y() << std::endl;
    std::cout << "radius = " << c.get_radius() << std::endl;
}
