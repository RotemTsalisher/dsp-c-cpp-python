#include <print>
#include <iostream>
#include <type_traits>

template <typename... Args>
requires (std::is_arithmetic_v<Args> && ...)
auto multiply_all(Args... args){
    
    return (args * ...);
}

int main()
{
    
    int (*fint)(int, int, int) = &multiply_all<int, int, int>;
    double (*fdouble)(double, double, double) = &multiply_all<double, double, double>;
    
    int res = fint(1,2,3);
    double res_ = fdouble(1.0, 1.5, 1.5);
    
    std::cout << "multiply_all(1,2,3)         = " << res << std::endl;
    std::cout << "multiply_all(1.0, 1.5, 1.5) = " << res_ << std::endl;
    return 0;
}
