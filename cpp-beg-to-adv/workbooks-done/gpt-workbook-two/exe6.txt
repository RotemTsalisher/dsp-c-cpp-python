#include <print>
#include <iostream>
#include <type_traits>

int apply_lambda(int a, int b, int (*f)(int, int)){
    return f(a, b);
};
int main()
{
    
    int (*subtract)(int, int) = [](int a, int b) {return b - a;};
    int a = 2, b = 10;
    
    std::cout << "res = " << apply_lambda(a,b,subtract) << std::endl;
    return 0;
}
