#include <print>
#include <iostream>
#include <thread>

static bool working = true;
static bool protecting = true;

void print_times(int value, int times){
    
    working = true;
    std::cout << "value: " << value << ", times: " << times <<std::endl;
    while(working){
        std::cout << "value: " << value << " || WORKING..." << std::endl;
        std::this_thread::sleep_for(std::chrono::seconds(1));  
    };
    
};

void free_timer(){
    
    while(protecting){
        std::cin.get();
        working = false;        
    }

}

int main()
{
    std::thread t0(free_timer);
    std::thread t1(print_times, 1, 20);
    std::thread t2(print_times, 2, 40);
    
    t1.join();
    t2.join();
    
    protecting = false;
    
    return 0;
}
