#include <iostream>

constexpr int square(int x) {
    return x * x;
};

constexpr double square(double x) {
    return x * x;
};

int main()
{
    static_assert(4 == square(2), "int math is broke!");
    static_assert(2.25 == square(1.5), "double math is broke!");
    
    static_assert(3 == square(2), "int fail example!");
    static_assert(1.0 == square(1.5), "double fail example!");
    return 0;
}
