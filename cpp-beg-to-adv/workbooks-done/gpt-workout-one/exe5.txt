#include <iostream>

template <typename... Args>
requires (std::is_arithmetic_v<Args> && ...)
auto sum_all(Args... args){
    return (args + ...);
};

int main()
{
    auto (*ptr)(int , int , int) = &sum_all<int, int, int>;
    auto (*ptr2)(double, double, double) = &sum_all<double, double, double>;
    
    std::cout << "sum_all(1,2,3) = " << ptr(1,2,3) << std::endl;
    std::cout << "sum_all(1.1,2.2,3.3) = " << ptr2(1.1,2.2,3.3) << std::endl;

    return 0;
}
