C++ Workbook 6

Exercise 1 — Function Overloading

Create two overloaded functions: square(int) square(double) Test both in main.
#include <iostream> int square(int x) { return x*x; } double square(double x) { return x*x; } int main() { std::cout << square(5) << std::endl; std::cout << square(2.5) << std::endl; }

Exercise 2 — Function Template

Create a template function: max_value(T a, T b) Return the larger value.
#include <iostream> template <typename T> T max_value(T a, T b) { if(a > b) return a; return b; } int main() { std::cout << max_value(5,3) << std::endl; std::cout << max_value(3.5,7.2) << std::endl; }

Exercise 3 — Template Specialization

Create template struct: type_multiplier Default value = 1 Specialize for double → value = 2 Create variable template: type_multiplier_v
#include <iostream> template <typename T> struct type_multiplier { static const int val = 1; }; template <> struct type_multiplier<double> { static const int val = 2; }; template <typename T> constexpr int type_multiplier_v = type_multiplier<T>::val; int main() { std::cout << type_multiplier_v<int> << std::endl; std::cout << type_multiplier_v<double> << std::endl; }

Exercise 4 — Lambda Function

Create a lambda that adds two numbers. Store it in a variable and call it.
#include <iostream> int main() { auto add = [](int a,int b) { return a + b; }; std::cout << add(3,4) << std::endl; }

Exercise 5 — Function Pointer

Create function: multiply(int,int) Store it in a function pointer and call it.
#include <iostream> int multiply(int a,int b) { return a*b; } int main() { int (*f)(int,int) = &multiply; std::cout << f(3,4) << std::endl; }

Exercise 6 — Passing Function as Argument

Create function: compute(int a,int b, function_pointer) Test with add and multiply.
#include <iostream> int add(int a,int b){ return a+b; } int multiply(int a,int b){ return a*b; } int compute(int a,int b,int (*f)(int,int)) { return f(a,b); } int main() { std::cout << compute(3,4,add) << std::endl; std::cout << compute(3,4,multiply) << std::endl; }

Exercise 7 — Fold Expression

Create function template: sum_all(args...) Use a fold expression.
#include <iostream> template <typename... Args> auto sum_all(Args... args) { return (args + ...); } int main() { std::cout << sum_all(1,2,3,4) << std::endl; }

Exercise 8 — Requires Constraint

Create function template: increment(T) Allow only integral types.
#include <iostream> #include <type_traits> template <typename T> requires std::is_integral_v<T> T increment(T v) { return v+1; } int main() { std::cout << increment(5) << std::endl; }

Exercise 9 — Namespace

Create namespace: math_utils Inside it create function: cube(int)
#include <iostream> namespace math_utils { int cube(int x) { return x*x*x; } } int main() { std::cout << math_utils::cube(3) << std::endl; }

Exercise 10 — Class With Encapsulation

Create class: Counter Private member: value Functions: set_value get_value increment
#include <iostream> class Counter { private: int value; public: void set_value(int v) { value = v; } int get_value() { return value; } void increment() { value++; } }; int main() { Counter c; c.set_value(5); c.increment(); std::cout << c.get_value() << std::endl; }

Exercise 11 — Thread Basics

Create a thread that prints numbers 1–5.
#include <iostream> #include <thread> void worker() { for(int i=1;i<=5;i++) { std::cout << i << std::endl; } } int main() { std::thread t(worker); t.join(); }

Exercise 12 — Class + Function Pointer

Create class Calculator. Add method: compute(int a,int b,function_pointer) Test with: • normal function • lambda
#include <iostream> class Calculator { public: int compute(int a,int b,int (*f)(int,int)) { return f(a,b); } }; int add(int a,int b) { return a+b; } int main() { Calculator c; auto mul = [](int a,int b) { return a*b; }; std::cout << c.compute(3,4,add) << std::endl; std::cout << c.compute(3,4,mul) << std::endl; }
