Exercise 1 — Struct vs Class

Create: • struct `Point` with public members • class `SafePoint` with private members Requirements: - struct allows direct access - class requires getters
#include struct Point { int x; int y; }; class SafePoint { private: int x; int y; public: SafePoint(int x_, int y_) { x = x_; y = y_; } int get_x(){ return x; } int get_y(){ return y; } }; int main() { Point p; p.x = 5; p.y = 6; SafePoint sp(5,6); std::cout << p.x << std::endl; std::cout << sp.get_x() << std::endl; }

Exercise 2 — Constructors

Create class `Rectangle`. Requirements: - private width and height - default constructor - constructor with parameters - function `area()`
#include class Rectangle { private: int width; int height; public: Rectangle() { width = 1; height = 1; } Rectangle(int w,int h) { width = w; height = h; } int area() { return width * height; } }; int main() { Rectangle r1; Rectangle r2(3,4); std::cout << r1.area() << std::endl; std::cout << r2.area() << std::endl; }

Exercise 3 — Class Object Through Pointer

Create class `Counter`. Requirements: - private value - increment function - getter Create object using a pointer.
#include class Counter { private: int value; public: Counter(int v) { value = v; } void increment() { value++; } int get() { return value; } }; int main() { Counter* c = new Counter(10); c->increment(); std::cout << c->get() << std::endl; delete c; }

Exercise 4 — Destructor

Create class `Tracker`. Requirements: - constructor prints "created" - destructor prints "destroyed" Create object inside a scope block.
#include class Tracker { public: Tracker() { std::cout << "created" << std::endl; } ~Tracker() { std::cout << "destroyed" << std::endl; } }; int main() { { Tracker t; } std::cout << "scope finished" << std::endl; }

Exercise 5 — this Pointer + Chaining

Create class `Accumulator`. Requirements: - private value - add(int) - multiply(int) Both functions return `this`. Allow: obj.add(3)->multiply(2)
#include class Accumulator { private: int value; public: Accumulator(int v) { value = v; } Accumulator* add(int v) { value += v; return this; } Accumulator* multiply(int v) { value *= v; return this; } int get() { return value; } }; int main() { Accumulator a(5); a.add(3)->multiply(2); std::cout << a.get() << std::endl; }

Exercise 6 — Classes Across Multiple Files + Include Guards

Split class `Car` across files: - Car.h - Car.cpp - main.cpp
// Car.h #ifndef CAR_H #define CAR_H class Car { private: int speed; public: Car(int s); int get_speed(); }; #endif // Car.cpp #include "Car.h" Car::Car(int s) { speed = s; } int Car::get_speed() { return speed; } // main.cpp #include #include "Car.h" int main() { Car c(120); std::cout << c.get_speed() << std::endl; }

Exercise 7 — Memory Alignment

Create: struct Data { char a; int b; char c; }; Print sizeof(Data). Explain the result.
#include struct Data { char a; int b; char c; }; int main() { std::cout << sizeof(Data) << std::endl; } Typical result: 12 bytes. Reason: char a -> 1 byte padding -> 3 bytes int b -> 4 bytes char c -> 1 byte padding -> 3 bytes Alignment inserted by compiler.
