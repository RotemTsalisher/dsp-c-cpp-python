#include <iostream>
#include <type_traits>
#include <thread>

static bool my_turn = true;

namespace worker {
    template <typename T>
    void run_task(T v){
        while(!my_turn);
        my_turn = false;
        for(size_t i = 0; i < 5; i ++){
            std::cout << "WORKING..." << std::endl;
            std::this_thread::sleep_for(std::chrono::seconds(1));
        }
        std::cout << "v = " << v << std::endl;
        my_turn = true;
    };
}

int main()
{
    std::thread w1(worker::run_task<int>, 10);
    std::thread w2(worker::run_task<int>, 20);
    
    w1.join();
    w2.join();
    return 0;
}
