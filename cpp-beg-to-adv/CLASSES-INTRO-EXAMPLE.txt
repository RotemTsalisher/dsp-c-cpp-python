#include <print>
#include <iostream>
#include <string>

class Dog {
    
    public: 
        std::string name;
        
    public:
        std::string bark() {return static_cast<std::string>("Woof!");};
        std::string greet() {return static_cast<std::string>("My name is " + name);};
        Dog() {name = "NoName";};
        Dog(std::string name_) { this->name = name_;};
        
};

int main()
{
    Dog snoopy;
    std::cout << "Dog's current name: " << std::endl;
    std::cout << snoopy.greet() << std::endl << std::endl;
    
    Dog mr_buckles("Mr Buckles");
    std::cout << "Dog's current name: " << std::endl;
    std::cout << mr_buckles.greet() << std::endl << std::endl;   
    
    snoopy.name = "Snoopy";
    
    std::cout << "Snoopy, BARK!" << std::endl;
    std::cout <<"Snoopy say: " << snoopy.bark() << std::endl << std::endl;
    
    std::cout << "Snoopy, Introduce yourself" << std::endl;
    std::cout << "Snoopy say: " << snoopy.greet() << std::endl;

    return 0;
}
