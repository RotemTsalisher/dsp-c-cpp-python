class Dog {
    private:
        std::string name, breed;
        int *p_age;
    public:
        Dog(){
            std::cout << "-- Empty Constructor --" << std::endl;
            name = "NULL";
            breed = "NULL";
            p_age = new int;
            *p_age = 0;
        };
        
        Dog(std::string name_, std::string breed_, int age) {
            std::cout << "-- Parameters Constructor --" << std::endl;
            name = name_;
            breed = breed_;
            p_age = new int;
            *p_age = age;
        };
        
        Dog(const Dog& d){
            std::cout << "-- COPY CONSTRUCTOR --" << std::endl;
            this->name = d.name;
            this->breed = d.breed;
            this->p_age = new int;
            *(this->p_age) = *(d.p_age);
        };
        
        ~Dog(){
            std::cout << "-- Destructor for Dog: {" << name << ", " << breed << ", " << *p_age << "}" << std::endl;
            delete p_age;
        };
};
